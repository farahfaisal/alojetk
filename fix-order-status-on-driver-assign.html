-- إنشاء دالة جديدة: تحويل الطلب إلى "قيد التحضير" عند تعيين السائق
CREATE OR REPLACE FUNCTION update_order_to_processing_on_driver_assign()
RETURNS TRIGGER
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- عند تعيين سائق جديد للطلب
  IF NEW.driver_id IS NOT NULL AND (OLD.driver_id IS NULL OR OLD.driver_id <> NEW.driver_id) THEN

    -- تحديث حالة الطلب إلى "قيد التحضير"
    UPDATE orders
    SET
      driver_id = NEW.driver_id,
      driver_name = NEW.driver_name,
      status = 'processing',  -- الحالة الجديدة: قيد التحضير
      updated_at = now()
    WHERE id = NEW.order_id
      AND status IN ('pending', 'confirmed');  -- فقط إذا كان في انتظار الموافقة أو مؤكد

    -- إنشاء سجل في تاريخ الحالات
    INSERT INTO order_status_history (
      order_id,
      status,
      note,
      created_by
    ) VALUES (
      NEW.order_id,
      'processing',
      'تم تعيين السائق ' || COALESCE(NEW.driver_name, 'غير معروف') || ' - الطلب قيد التحضير',
      NEW.driver_id
    );

  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- إنشاء trigger جديد
CREATE TRIGGER assign_driver_to_processing_trigger
  AFTER UPDATE OF driver_id ON driver_waiting_list
  FOR EACH ROW
  WHEN (NEW.driver_id IS NOT NULL)
  EXECUTE FUNCTION update_order_to_processing_on_driver_assign();

-- تعليق توضيحي
COMMENT ON FUNCTION update_order_to_processing_on_driver_assign() IS 'عند تعيين سائق، يتحول الطلب إلى قيد التحضير (وليس في الطريق)';
COMMENT ON FUNCTION start_delivery_trip IS 'السائق يستخدمها يدوياً عند البدء الفعلي للرحلة - تحول الطلب إلى في الطريق';
COMMENT ON FUNCTION start_multi_delivery_trip IS 'السائق يستخدمها يدوياً عند بدء رحلة توصيل متعددة';
